/* =========================================================================
   PULSELAB APPLICATION ROOT COMPONENT
   ========================================================================= */

export default function App() {
  const [currentPage, setCurrentPage] = useState<PageView>('home');
  const [selectedGuideId, setSelectedGuideId] = useState<string | null>(null);
  const [searchQuery, setSearchQuery] = useState('');
  const [isMobileMenuOpen, setIsMobileMenuOpen] = useState(false);

  // Navigate helper
  const navigateTo = (page: PageView, guideId?: string) => {
    setCurrentPage(page);
    if (guideId) {
      setSelectedGuideId(guideId);
    }
    setIsMobileMenuOpen(false);
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  return (
    <div className="min-h-screen bg-slate-50 text-slate-900 flex flex-col font-sans antialiased selection:bg-teal-500 selection:text-white">
      {/* 1. TOP EMERGENCY BAR (Always visible for safety) */}
      <EmergencyTopBanner onEmergencyClick={() => navigateTo('emergency')} />

      {/* 2. MAIN NAVIGATION */}
      <Header
        currentPage={currentPage}
        onNavigate={navigateTo}
        isMobileMenuOpen={isMobileMenuOpen}
        setIsMobileMenuOpen={setIsMobileMenuOpen}
      />

      {/* 3. MAIN ROUTED VIEW */}
      <main className="flex-1">
        {currentPage === 'home' && (
          <HomePage
            onNavigate={navigateTo}
            onSelectGuide={(id) => navigateTo('guides', id)}
          />
        )}
        {currentPage === 'guides' && (
          <HealthGuidesPage
            initialGuideId={selectedGuideId}
            onNavigate={navigateTo}
          />
        )}
        {currentPage === 'hospitals' && <HospitalsPage />}
        {currentPage === 'emergency' && <EmergencyPage onNavigate={navigateTo} />}
        {currentPage === 'assistant' && (
          <AssistantPage
            onNavigate={navigateTo}
            onSelectGuide={(id) => navigateTo('guides', id)}
          />
        )}
        {currentPage === 'about' && <AboutPage onNavigate={navigateTo} />}
        {currentPage === 'contact' && <ContactPage />}
      </main>

      {/* 4. FOOTER & CLINICAL DISCLAIMER */}
      <Footer onNavigate={navigateTo} />
    </div>
  );
}

// Great-circle Haversine formula (km)
function calculateDistanceKm(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const R = 6371; // Earth radius in km
  const dLat = ((lat2 - lat1) * Math.PI) / 180;
  const dLon = ((lon2 - lon1) * Math.PI) / 180;
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos((lat1 * Math.PI) / 180) *
      Math.cos((lat2 * Math.PI) / 180) *
      Math.sin(dLon / 2) *
      Math.sin(dLon / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return Math.round(R * c * 10) / 10;
}

  // Request browser geolocation
  const handleDetectLocation = () => {
    if (!navigator.geolocation) {
      setLocationError('Geolocation is not supported by your browser.');
      return;
    }

    setIsLoadingLocation(true);
    setLocationError(null);

    navigator.geolocation.getCurrentPosition(
      async (pos) => {
        const lat = pos.coords.latitude;
        const lon = pos.coords.longitude;
        setUserCoords({ lat, lon });
        setIsLoadingLocation(false);
        // Query live OpenStreetMap Overpass API for real amenities near coordinates
        fetchLiveNearbyFacilities(lat, lon);
      },
      (err) => {
        setIsLoadingLocation(false);
        if (err.code === err.PERMISSION_DENIED) {
          setLocationError('Location permission was denied. You can search manually by Indian PIN code or City below.');
        } else {
          setLocationError('Unable to detect location. Please type your city or locality manually.');
        }
      },
      { timeout: 12000, enableHighAccuracy: true }
    );
  };

    setIsFetchingOverpass(true);
    setLiveDataSource(null);

      [out:json][timeout:15];
      (
        node["amenity"="hospital"](around:8000,${lat},${lon});
        node["amenity"="clinic"](around:8000,${lat},${lon});
        node["amenity"="pharmacy"](around:8000,${lat},${lon});
      );
      out body 15;
    `;

    try {
      const resp = await fetch('https://overpass-api.de/api/interpreter', {
        method: 'POST',

                    <a
                      href={`tel:${facility.phone.replace(/[^0-9]/g, '')}`}
                      className="flex-1 bg-teal-50 hover:bg-teal-100 text-teal-800 font-bold text-xs py-2 px-3 rounded-lg flex items-center justify-center gap-1.5 transition"
                    >
                      <Phone className="w-3.5 h-3.5 text-teal-600" /> Call
                    </a>
                  ) : (
                    <span className="flex-1 text-center text-[11px] text-slate-400 py-2">
                      Phone on Map
                    </span>
                  )}

                  <a
                    href={mapsUrl}
                    target="_blank"
                    rel="noopener noreferrer"
                    className="flex-1 bg-slate-900 hover:bg-slate-800 text-white font-bold text-xs py-2 px-3 rounded-lg flex items-center justify-center gap-1.5 transition"
                  >
                    <ExternalLink className="w-3.5 h-3.5 text-teal-400" /> Directions
                  </a>
                </div>
              </div>
            );
          })}
        </div>
      ) : (
        <div className="bg-white rounded-3xl border border-slate-200 p-10 text-center space-y-3">
          <MapPin className="w-10 h-10 text-slate-300 mx-auto" />
          <h3 className="text-base font-bold text-slate-800">No medical facilities found for this filter</h3>
          <p className="text-xs text-slate-500 max-w-md mx-auto">
            Try switching to &ldquo;All Facilities&rdquo; or tap below to search live Google Maps for facilities in your locality.
          </p>
          <a
            href={getManualSearchMapsLink()}
            target="_blank"
            rel="noopener noreferrer"
            className="inline-flex items-center gap-2 bg-teal-600 text-white text-xs font-bold px-4 py-2 rounded-xl"
          >
            <ExternalLink className="w-4 h-4" /> Open Google Maps Live Directory
          </a>
        </div>
      )}

      {/* TECHNICAL BACKEND CONFIGURATION NOTE */}
      <div className="bg-slate-100 border border-slate-200 rounded-2xl p-4 text-xs text-slate-600 space-y-1">
        <p className="font-bold text-slate-800 flex items-center gap-1.5">
          <Shield className="w-4 h-4 text-teal-600" /> Production API Integration Notice
        </p>
        <p>
          PulseLab uses OpenStreetMap Overpass and Google Maps deep links. In production, configure your secure server-side proxy with the <code>GOOGLE_MAPS_PLACES_API_KEY</code> environment variable to stream live verified bed availability and real-time hospital triage capacity without exposing credentials in client-side code.
        </p>
      </div>
    </div>
  );
}

/* =========================================================================
   PULSELAB HEALTH ASSISTANT (CONVERSATIONAL SYMPTOM TRIAGE)
   ========================================================================= */

interface Message {
  id: string;
  sender: 'bot' | 'user';
  text: string;
  isEmergencyAlert?: boolean;
  suggestedGuideId?: string;
  quickReplies?: string[];
}

function AssistantPage({
  onNavigate,
  onSelectGuide
}: {
  onNavigate: (p: PageView) => void;
  onSelectGuide: (id: string) => void;
}) {
  const [messages, setMessages] = useState<Message[]>([
    {
      id: 'welcome-1',
      sender: 'bot',
      text: 'Namaste! I am the PulseLab Health Assistant. Describe your symptoms in everyday language (e.g. "I have had a high fever and headache since yesterday" or "My chest feels tight").',
      quickReplies: ['Fever & Sore throat', 'Chest tightness', 'Sudden dizziness', 'Diarrhea & Vomiting']
    }
  ]);
  const [inputValue, setInputValue] = useState('');
  const messagesEndRef = useRef<HTMLDivElement>(null);

  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  useEffect(() => {
    scrollToBottom();
  }, [messages]);

  const handleSend = (textToSend?: string) => {
    const text = (textToSend || inputValue).trim();
    if (!text) return;

    const userMsg: Message = {
      id: `user-${Date.now()}`,
      sender: 'user',
      text
    };

    setMessages((prev) => [...prev, userMsg]);
    if (!textToSend) setInputValue('');

    // Process Clinical Decision Rules
    setTimeout(() => {
      processTriageResponse(text.toLowerCase());
    }, 450);
  };

  const processTriageResponse = (q: string) => {
    // 1. Critical Emergency Rules (Immediate priority)
    const isCardiac = q.includes('chest') || q.includes('heart') || q.includes('arm pain') || q.includes('crushing');
    const isStroke = q.includes('stroke') || q.includes('face drop') || q.includes('speech') || q.includes('slurring') || q.includes('paralysis');
    const isPoisoning = q.includes('poison') || q.includes('overdose') || q.includes('chemical') || q.includes('rat poison');
    const isSevereBreathing = q.includes('cannot breathe') || q.includes('choking') || q.includes('blue lips') || q.includes('severe asthma');

    if (isCardiac) {
      setMessages((prev) => [
        ...prev,
        {
          id: `bot-${Date.now()}`,
          sender: 'bot',
          isEmergencyAlert: true,
          text: 'CRITICAL ALERT: Chest discomfort, pressure, or radiating pain can indicate an acute heart attack. Do NOT wait for symptoms to improve.',
          suggestedGuideId: 'chest-pain-heart',
          quickReplies: ['Call 112 Right Now', 'What to do while waiting']
        }
      ]);
      return;
    }

    if (isStroke) {
      setMessages((prev) => [
        ...prev,
        {
          id: `bot-${Date.now()}`,
          sender: 'bot',
          isEmergencyAlert: true,
          text: 'EMERGENCY: Sudden facial drooping, arm numbness, or speech slurring are clinical signs of Acute Stroke (F.A.S.T.). Rapid intervention within minutes is essential.',
          suggestedGuideId: 'stroke-fast',
          quickReplies: ['Call 112 Immediately', 'View FAST Guide']
        }
      ]);
      return;
    }

    if (isPoisoning) {
      setMessages((prev) => [
        ...prev,
        {
          id: `bot-${Date.now()}`,
          sender: 'bot',
          isEmergencyAlert: true,
          text: 'URGENT: For suspected poisoning or chemical ingestion, never induce vomiting. Call 112 and keep the product container for medical personnel.',
          suggestedGuideId: 'other-emergencies'
        }
      ]);
      return;
    }

    if (isSevereBreathing) {
      setMessages((prev) => [
        ...prev,
        {
          id: `bot-${Date.now()}`,
          sender: 'bot',
          isEmergencyAlert: true,
          text: 'URGENT: Severe breathing distress requires immediate emergency treatment. Sit upright and avoid lying down.',
          suggestedGuideId: 'breathing-allergies'
        }
      ]);
      return;
    }

    // 2. Secondary Health Categories
    if (q.includes('fever') || q.includes('temperature') || q.includes('cold') || q.includes('cough') || q.includes('throat')) {
      setMessages((prev) => [
        ...prev,
        {
          id: `bot-${Date.now()}`,
          sender: 'bot',
          text: 'This appears related to Fever & Common Respiratory Infections. For most mild viral infections, rest, hydration (ORS/fluids), and fever monitoring are safe primary steps. Watch for red flags like fever lasting >3 days, severe breathlessness, or rashes.',
          suggestedGuideId: 'fever-infections',
          quickReplies: ['When to see a doctor?', 'Find nearby clinics']
        }
      ]);
      return;
    }

    if (q.includes('stomach') || q.includes('vomit') || q.includes('diarrhea') || q.includes('loose motion') || q.includes('cramp')) {
      setMessages((prev) => [
        ...prev,
        {
          id: `bot-${Date.now()}`,
          sender: 'bot',
          text: 'This is consistent with Stomach Illness & Dehydration. Sip WHO-standard Oral Rehydration Salts (ORS) slowly in small quantities. Seek clinical evaluation if unable to keep liquids down, experiencing black stools, or severe localized abdominal pain.',
          suggestedGuideId: 'stomach-dehydration',
          quickReplies: ['ORS instructions', 'Warning signs']
        }
      ]);
      return;
    }

    if (q.includes('burn') || q.includes('cut') || q.includes('wound') || q.includes('bleeding') || q.includes('injury') || q.includes('sprain')) {
      setMessages((prev) => [
        ...prev,
        {
          id: `bot-${Date.now()}`,
          sender: 'bot',
          text: 'For burns: cool under running water for 15 minutes (no toothpaste or ice). For wounds: apply continuous direct pressure with a clean cloth. If bleeding does not stop after 10 minutes, seek emergency trauma care.',
          suggestedGuideId: 'injuries-burns-bleeding'
        }
      ]);
      return;
    }

    // Default general response
    setMessages((prev) => [
      ...prev,
      {
        id: `bot-${Date.now()}`,
        sender: 'bot',
        text: 'Thank you for sharing your concerns. PulseLab does not provide clinical diagnoses or individualized prescriptions. Please select one of our 8 comprehensive guides or locate a nearby qualified clinic for an in-person assessment.',
        quickReplies: ['View all 8 Guides', 'Find nearest clinic']
      }
    ]);
  };

  return (
    <div className="max-w-4xl mx-auto px-4 sm:px-6 lg:px-8 py-10 space-y-6">
      {/* HEADER */}
      <div className="border-b border-slate-200 pb-4">
        <span className="text-teal-600 font-bold uppercase tracking-wider text-xs">Conversational Triage</span>
        <h1 className="text-2xl sm:text-3xl font-extrabold text-slate-900 mt-1">
          PulseLab Health Assistant
        </h1>
        <p className="text-xs sm:text-sm text-slate-600 mt-1">
          Enter plain-language symptoms. The assistant detects danger indicators and matches your situation to appropriate care pathways.
        </p>
      </div>

      {/* CHAT CONTAINER */}
      <div className="bg-white border border-slate-200 rounded-3xl shadow-sm flex flex-col h-[540px] overflow-hidden">
        {/* Messages Feed */}
        <div className="flex-1 p-4 sm:p-6 overflow-y-auto space-y-4 bg-slate-50/50">
          {messages.map((m) => {
            const isBot = m.sender === 'bot';
            return (
              <div
                key={m.id}
                className={`flex flex-col ${isBot ? 'items-start' : 'items-end'}`}
              >
                <div
                  className={`max-w-[85%] sm:max-w-[75%] rounded-2xl px-4 py-3 text-sm leading-relaxed shadow-sm ${
                    isBot
                      ? m.isEmergencyAlert
                        ? 'bg-red-50 border border-red-200 text-red-950 font-medium'
                        : 'bg-white border border-slate-200 text-slate-800'
                      : 'bg-teal-600 text-white font-medium'
                  }`}
                >
                  {m.isEmergencyAlert && (
                    <div className="flex items-center gap-1.5 text-xs font-bold text-red-700 uppercase tracking-wide mb-1">
                      <AlertTriangle className="w-4 h-4 text-red-600" /> Urgent Care Alert
                    </div>
                  )}

                  <p>{m.text}</p>

                  {/* Red Alert Call Button in chat */}
                  {m.isEmergencyAlert && (
                    <div className="mt-3 pt-2 border-t border-red-200 flex flex-wrap gap-2">
                      <a
                        href={`tel:${EMERGENCY_NUMBER}`}
                        className="bg-red-600 hover:bg-red-700 text-white text-xs font-bold px-3 py-1.5 rounded-lg inline-flex items-center gap-1.5"
                      >
                        <Phone className="w-3.5 h-3.5" /> Call 112 Right Now
                      </a>
                    </div>
                  )}

                  {/* Suggestion button */}
                  {m.suggestedGuideId && (
                    <div className="mt-3 pt-2 border-t border-slate-100 flex items-center justify-between">
                      <button
                        onClick={() => onSelectGuide(m.suggestedGuideId!)}
                        className="text-xs font-bold text-teal-700 hover:underline flex items-center gap-1"
                      >
                        Open Relevant Health Guide <ChevronRight className="w-3.5 h-3.5" />
                      </button>
                    </div>
                  )}
                </div>

                {/* Quick replies */}
                {m.quickReplies && (
                  <div className="flex flex-wrap gap-1.5 mt-2">
                    {m.quickReplies.map((qr, i) => (
                      <button
                        key={i}
                        onClick={() => {
                          if (qr === 'Call 112 Right Now' || qr === 'Call 112 Immediately') {
                            window.location.href = `tel:${EMERGENCY_NUMBER}`;
                          } else if (qr === 'Find nearby clinics' || qr === 'Find nearest clinic') {
                            onNavigate('hospitals');
                          } else if (qr === 'View all 8 Guides') {
                            onNavigate('guides');
                          } else {
                            handleSend(qr);
                          }
                        }}
                        className="bg-white border border-teal-200 text-teal-800 hover:bg-teal-50 text-xs px-2.5 py-1 rounded-full shadow-2xs transition"
                      >
                        {qr}
                      </button>
                    ))}
                  </div>
                )}
              </div>
            );
          })}
          <div ref={messagesEndRef} />
        </div>

        {/* INPUT BOX */}
        <div className="p-3 bg-white border-t border-slate-200 flex items-center gap-2">
          <input
            type="text"
            value={inputValue}
            onChange={(e) => setInputValue(e.target.value)}
            onKeyDown={(e) => {
              if (e.key === 'Enter') handleSend();
            }}
            placeholder="Type your health concern or symptoms here..."
            className="flex-1 border border-slate-200 rounded-xl px-4 py-2.5 text-sm outline-none focus:ring-2 focus:ring-teal-400 bg-slate-50"
            aria-label="Enter your symptom"
          />
          <button
            onClick={() => handleSend()}
            disabled={!inputValue.trim()}
            className="bg-teal-600 hover:bg-teal-700 text-white p-2.5 rounded-xl disabled:opacity-40 transition"
            aria-label="Send message"
          >
            <Send className="w-5 h-5" />
          </button>
        </div>
      </div>

      <div className="text-center text-xs text-slate-500 max-w-xl mx-auto leading-relaxed">
        <strong>Safety Note:</strong> PulseLab Health Assistant does not prescribe medicine or provide conclusive diagnoses. If you experience severe pain, difficulty breathing, or sudden weakness, call 112 without delay.
      </div>
    </div>
  );
}

   EMERGENCY HELP PAGE (NATIONAL DISPATCH & IMMEDIATE LIFE-SAVING PROTOCOLS)
   ========================================================================= */

function EmergencyPage({ onNavigate }: { onNavigate: (p: PageView) => void }) {
  return (
    <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-10 space-y-12">
      {/* HERO ALERT */}
      <div className="bg-gradient-to-r from-red-700 via-red-600 to-rose-700 text-white rounded-3xl p-6 sm:p-10 shadow-2xl space-y-6">
        <div className="flex flex-col md:flex-row items-start md:items-center justify-between gap-6">
          <div className="space-y-2 max-w-2xl">
            <span className="bg-white/20 text-white font-bold text-xs uppercase px-3 py-1 rounded-full inline-block">
              India Unified Emergency Service
            </span>
            <h1 className="text-3xl sm:text-5xl font-black tracking-tight">
              Emergency Medical Helpline: 112
            </h1>
